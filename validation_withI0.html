# SIR simulation model (odin2)
simple_sir <- odin2::odin({
  N      <- parameter()
  I0     <- parameter()
  gamma  <- parameter()
  n_beta <- parameter()
  beta_times  <- parameter()
  beta_values <- parameter()
  dim(beta_times)  <- n_beta
  dim(beta_values) <- n_beta
  beta_t <- interpolate(beta_times, beta_values, "constant")
  initial(S) <- N - I0
  initial(I) <- I0
  initial(incidence, zero_every = 1) <- 0
  p_SI <- 1 - exp(-beta_t * I / N * dt)
  p_IR <- 1 - exp(-gamma * dt)
  n_SI <- S * p_SI
  n_IR <- I * p_IR
  update(S) <- S - n_SI
  update(I) <- I + n_SI - n_IR
  update(incidence) <- n_SI
  cases <- data()
  cases ~ Poisson(incidence)
})

simulate_epidemic <- function(N, gamma, I0, beta_values, beta_times,
                              T = 150, seed = 444) {
  set.seed(seed)
  obj  <- simple_sir()
  pars <- list(N = N, I0 = I0, gamma = gamma,
               n_beta = length(beta_values),
               beta_times = beta_times, beta_values = beta_values)
  sys <- dust2::dust_system_create(obj, pars, n_particles = 1, dt = 1)
  dust2::dust_system_set_state_initial(sys)
  y <- dust2::dust_system_simulate(sys, 0:T)
  list(
    incidence_df = data.frame(time  = 1:T,
                              cases = rpois(T, y[3, -1])),
    S_true       = y[1, -1],
    inc_true     = y[3, -1]
  )
}

compute_true_Rt <- function(beta_values, beta_times, gamma,
                            S_true, N, T = 150) {
  beta_t <- approx(beta_times + 1, beta_values, 1:T,
                   method = "constant", f = 0, rule = 2)$y
  (beta_t / gamma) * (S_true / N)
}

fit_scenario <- function(incidence_df, N, gamma, beta_breaks,
                         mean_beta, sd_beta = 0.7,
                         rw_sd_beta = 0.2,
                         mean_I0, sd_I0 = 0.5,
                         n_steps = 15000, n_chains = 3) {
  estimate_R_blocks(
    data        = incidence_df,
    N           = N,
    gamma       = gamma,
    beta_breaks = beta_breaks,
    priors = list(mean_beta  = mean_beta, sd_beta    = sd_beta,
                  rw_sd_beta = rw_sd_beta,
                  mean_I0    = mean_I0,   sd_I0      = sd_I0),
    inits  = list(beta = exp(mean_beta),  I0 = exp(mean_I0)),
    mcmc   = list(n_steps = n_steps, burnin = 0.5,
                  seed = 444, n_chains = n_chains)
  )
}

coverage_stats <- function(Rt_true, lwr, upr, med,
                           scenario, label = "") {
  n       <- min(length(Rt_true), length(lwr), length(upr), length(med))
  covered <- Rt_true[1:n] >= lwr[1:n] & Rt_true[1:n] <= upr[1:n]
  data.frame(
    Scenario  = label,
    `Coverage (%)` = round(mean(covered) * 100, 1),
    `Mean bias`    = round(mean(med[1:n] - Rt_true[1:n]), 3),
    RMSE           = round(sqrt(mean((med[1:n] - Rt_true[1:n])^2)), 3),
    `Mean CI width`= round(mean(upr[1:n] - lwr[1:n]), 3),
    check.names = FALSE
  )
}

# ---- Helper: extract blockwise beta(t) summary from beta_t_mat ----
extract_beta_blocks <- function(out) {
  beta_mat <- out$beta_t_mat
  bt       <- out$beta_times
  time_vec <- out$Rt_quantiles$time
  K        <- length(bt)
  rows <- lapply(seq_len(K), function(k) {
    t_start <- bt[k]
    t_end   <- if (k < K) bt[k + 1] - 1 else max(time_vec)
    idx     <- which(time_vec >= t_start & time_vec <= t_end)
    draws   <- beta_mat[idx, , drop = FALSE]
    data.frame(t_start = t_start, t_end = t_end,
               median = median(draws),
               lower  = quantile(draws, 0.025),
               upper  = quantile(draws, 0.975))
  })
  do.call(rbind, rows)
}

# ---- Helper: EpiEstim block summary ----
epi_block_summary <- function(res) {
  res$R |>
    transmute(t_start, t_end,
              Mean  = `Mean(R)`,
              Lower = `Quantile.0.025(R)`,
              Upper = `Quantile.0.975(R)`)
}

# ---- Helper: rodent Rt_mat block summary *by user breaks* ----
rodent_block_summary_by_breaks <- function(out, breaks) {
  Rt_mat   <- out$Rt_mat
  time_vec <- out$Rt_quantiles$time
  K        <- length(breaks)
  rows <- lapply(seq_len(K), function(k) {
    t_start <- breaks[k]
    t_end   <- if (k < K) breaks[k + 1] - 1L else max(time_vec)
    idx     <- which(time_vec >= t_start & time_vec <= t_end)
    draws   <- Rt_mat[idx, , drop = FALSE]
    data.frame(
      t_start = t_start, t_end = t_end,
      Mean    = mean(draws),
      Lower   = as.numeric(quantile(draws, 0.025)),
      Upper   = as.numeric(quantile(draws, 0.975))
    )
  })
  do.call(rbind, rows)
}
