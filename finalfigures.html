# --- S(t) trajectory plot ---
plot_S_trajectory <- function(out, N, title_label = "") {
  df <- out$S_quantiles
  ggplot(df, aes(x = time)) +
    geom_ribbon(aes(ymin = lwr, ymax = upr),
                fill = "#56B4E9", alpha = 0.3) +
    geom_line(aes(y = median), colour = "#56B4E9", linewidth = 1.1) +
    labs(
      x        = "Time (days)",
      y        = "S(t) — susceptible individuals",
      title    = paste0(title_label, " — Posterior S(t)"),
      subtitle = paste0("Posterior median and 95% CI, N = ",
                        format(N, big.mark = ","))
    ) +
    theme_minimal(base_size = 13)
}

# --- Smooth Rt(t) plot ---
plot_Rt <- function(out, title_label = "Rt(t)", vlines = NULL) {
  df <- out$Rt_quantiles
  p <- ggplot(df, aes(x = time)) +
    geom_ribbon(aes(ymin = lwr, ymax = upr),
                fill = "#0072B2", alpha = 0.25) +
    geom_line(aes(y = median), colour = "#0072B2", linewidth = 1.1) +
    geom_hline(yintercept = 1, linetype = 2, colour = "red") +
    labs(x = "Time (days)", y = "Rt",
         title    = title_label,
         subtitle = "Posterior median and 95% CI") +
    theme_minimal(base_size = 13)
  if (!is.null(vlines))
    p <- p + geom_vline(xintercept = vlines, linetype = "dotted",
                        color = "darkgrey", linewidth = 0.7)
  p
}

# --- Block Rt from Rt_mat ---
block_Rt <- function(Rt_mat, beta_times, time_vec) {
  K   <- length(beta_times)
  out <- data.frame()
  for (k in seq_len(K)) {
    t_start     <- beta_times[k]
    t_end       <- if (k < K) beta_times[k + 1] - 1 else max(time_vec)
    idx         <- which(time_vec >= t_start & time_vec <= t_end)
    block_draws <- Rt_mat[idx, , drop = FALSE]
    out <- rbind(out, data.frame(
      t_start = t_start,
      t_end   = t_end,
      Mean    = mean(block_draws),
      Lower   = quantile(block_draws, 0.025),
      Upper   = quantile(block_draws, 0.975)
    ))
  }
  out
}

# --- Block Rt plot ---
plot_block_Rt <- function(df_blocks, title_label = "", vlines = NULL) {
  p <- ggplot(df_blocks) +

              fill = "#0072B2", alpha = 0.25, color = NA) +

                 color = "#0072B2", linewidth = 1.2) +
    geom_hline(yintercept = 1, linetype = 2, color = "red") +
    labs(x = "Time (days)", y = "Rt",
         title    = paste0(title_label, " — rodent: Block Rt"),

    p <- p + geom_vline(xintercept = vlines, linetype = "dotted",
                        color = "darkgrey", linewidth = 0.7)
  p
}

# --- beta(t) block plot from Rt_mat / beta_t_mat ---
plot_beta_blocks <- function(out, vlines = NULL, title_label = "") {
  beta_mat <- out$beta_t_mat
  bt       <- out$beta_times
  time_vec <- out$Rt_quantiles$time
  K        <- length(bt)

  rows <- lapply(seq_len(K), function(k) {
    t_start <- bt[k]
    t_end   <- if (k < K) bt[k + 1] - 1 else max(time_vec)
    idx     <- which(time_vec >= t_start & time_vec <= t_end)
    draws   <- beta_mat[idx, , drop = FALSE]
    data.frame(
      t_start = t_start,
      t_end   = t_end,
      median  = median(draws),
      lower   = quantile(draws, 0.025),
      upper   = quantile(draws, 0.975)
    )
  })
  df <- do.call(rbind, rows)

  p <- ggplot(df) +
    geom_rect(aes(xmin = t_start, xmax = t_end, ymin = lower, ymax = upper),
              fill = "#6A51A3", alpha = 0.15, color = NA) +
    geom_segment(aes(x = t_start, xend = t_end, y = median, yend = median),
                 color = "#6A51A3", linewidth = 0.9, lineend = "butt") +
    labs(x = "Time (days)", y = expression(beta(t)),
         title    = paste0(title_label, " — rodent: β(t) blockwise"),
         subtitle = "Posterior median and 95% CI by block") +
    theme_minimal(base_size = 13) +
    theme(panel.grid = element_blank())
  if (!is.null(vlines))
    p <- p + geom_vline(xintercept = vlines, linetype = "dotted",
                        color = "darkgrey", linewidth = 0.6)
  p
}

# --- rodent vs EpiEstim comparison plot ---
make_comparison_plot <- function(df_epi, df_rod, vlines = NULL,
                                 title_label = "") {
  span <- max(df_rod$t_end) - min(df_rod$t_start)
  df <- bind_rows(
    df_epi %>% mutate(method = "EpiEstim"),
    df_rod %>% mutate(method = "rodent")
  ) %>%
    mutate(
      x_start = ifelse(method == "rodent",
                       t_start + 0.02 * span, t_start - 0.02 * span),
      x_end   = ifelse(method == "rodent",
                       t_end   + 0.02 * span, t_end   - 0.02 * span)
    )
  p <- ggplot(df) +
    geom_rect(aes(xmin = x_start, xmax = x_end,
                  ymin = Lower, ymax = Upper, fill = method),
              alpha = 0.3, color = NA) +
    geom_segment(aes(x = x_start, xend = x_end,
                     y = Mean, yend = Mean, color = method),
                 linewidth = 1.2) +
    geom_hline(yintercept = 1, linetype = 2, color = "black") +
    scale_fill_manual(values  = c("EpiEstim" = "#E69F00",
                                  "rodent"   = "#0072B2")) +
    scale_color_manual(values = c("EpiEstim" = "#E69F00",
                                  "rodent"   = "#0072B2")) +
    labs(x = "Time (days)", y = "Rt",
         title    = paste0(title_label, " — Block Rt comparison"),
         subtitle = "Orange = EpiEstim, Blue = rodent",
         fill = NULL, color = NULL) +
    theme_minimal(base_size = 13) +
    theme(legend.position = "bottom")
  if (!is.null(vlines))
    p <- p + geom_vline(xintercept = vlines, linetype = "dotted",
                        color = "darkgrey", linewidth = 0.6)
  p
}

# --- numeric summary printer ---
print_Rt_comparison <- function(df_epi, df_rod, label = "") {
  cat("\n=====", label, "— Rt COMPARISON =====\n\n")
  for (i in seq_len(nrow(df_epi))) {
    cat(sprintf("Block %d (days %d-%d):\n", i,
                df_epi$t_start[i], df_epi$t_end[i]))
    cat(sprintf("  EpiEstim : %.2f [%.2f, %.2f]\n",
                df_epi$Mean[i], df_epi$Lower[i], df_epi$Upper[i]))
    cat(sprintf("  rodent   : %.2f [%.2f, %.2f]\n\n",
                df_rod$Mean[i], df_rod$Lower[i], df_rod$Upper[i]))
  }
}

simple_sir <- odin2::odin({
  N      <- parameter()
  I0     <- parameter()
  gamma  <- parameter()
  n_beta <- parameter()
  beta_times  <- parameter()
  beta_values <- parameter()
  dim(beta_times)  <- n_beta
  dim(beta_values) <- n_beta
  beta_t <- interpolate(beta_times, beta_values, "constant")
  initial(S) <- N - I0
  initial(I) <- I0
  initial(incidence, zero_every = 1) <- 0
  p_SI <- 1 - exp(-beta_t * I / N * dt)
  p_IR <- 1 - exp(-gamma * dt)
  n_SI <- S * p_SI
  n_IR <- I * p_IR
  update(S) <- S - n_SI
  update(I) <- I + n_SI - n_IR
  update(incidence) <- n_SI
  cases <- data()
  cases ~ Poisson(incidence)
})

simulate_epidemic <- function(N, gamma, I0, beta_values, beta_times,
                              T = 150, seed = 444) {
  set.seed(seed)
  obj  <- simple_sir()
  pars <- list(N = N, I0 = I0, gamma = gamma,
               n_beta      = length(beta_values),
               beta_times  = beta_times,
               beta_values = beta_values)
  sys <- dust2::dust_system_create(obj, pars, n_particles = 1, dt = 1)
  dust2::dust_system_set_state_initial(sys)
  y <- dust2::dust_system_simulate(sys, 0:T)
  list(
    incidence_df = data.frame(time  = 1:T,
                              cases = rpois(T, y[3, -1])),
    S_true       = y[1, -1],
    inc_true     = y[3, -1]
  )
}

compute_true_Rt <- function(beta_values, beta_times, gamma,
                            S_true, N, T = 150) {
  beta_t <- approx(beta_times + 1, beta_values, 1:T,
                   method = "constant", f = 0, rule = 2)$y
  (beta_t / gamma) * (S_true / N)
}

coverage_stats <- function(Rt_true, lwr, upr, med,
                           scenario, label = "") {
  n       <- min(length(Rt_true), length(lwr), length(upr), length(med))
  covered <- Rt_true[1:n] >= lwr[1:n] & Rt_true[1:n] <= upr[1:n]
  data.frame(
    label     = label,
    scenario  = scenario,
    coverage  = round(mean(covered) * 100, 1),
    mean_bias = round(mean(med[1:n] - Rt_true[1:n]), 3),
    rmse      = round(sqrt(mean((med[1:n] - Rt_true[1:n])^2)), 3),
    ci_width  = round(mean(upr[1:n] - lwr[1:n]), 3)
  )
}
